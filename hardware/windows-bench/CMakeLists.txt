cmake_minimum_required(VERSION 3.16)
project(mcl_windows_bench C)
set(MCL_SDK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MCL_AP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MCL_AP_BUILD_EXPERIMENTS OFF CACHE BOOL "" FORCE)
get_filename_component(MCL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${MCL_ROOT}/mcl-sdk" sdk)
add_subdirectory("${MCL_ROOT}/mcl-ap" ap)
add_library(mcl_windows_bench SHARED bench.c "${MCL_ROOT}/mcl-ble/src/ble_binding.c")
target_include_directories(mcl_windows_bench PRIVATE "${MCL_ROOT}/mcl-ble/include")
target_link_libraries(mcl_windows_bench PRIVATE mcl_sdk mcl_ap_listen)
set_target_properties(mcl_windows_bench PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(MSVC)
  target_compile_options(mcl_windows_bench PRIVATE /W4 /WX)
endif()
